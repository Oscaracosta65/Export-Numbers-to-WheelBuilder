function parseCsvNumbers(raw){
  var out = [];
  var seen = {};
  var parts = String(raw || '').split(',');
  for (var i = 0; i < parts.length; i++){
    var p = String(parts[i] || '').trim();
    if (!/^\-?\d+$/.test(p)) { continue; }
    var n = parseInt(p, 10);
    if (!(typeof n === 'number' && isFinite(n))) { continue; }
    if (n < 0) { continue; }
    if (!seen[n]){
      seen[n] = true;
      out.push(n);
    }
  }
  return out;
}

    var n = toPositiveInt(mains[i]);
    if (n === 0) { hasZero = true; cleaned.push(0); continue; }
    if (n > 0) { cleaned.push(n); if (n > maxSeen) { maxSeen = n; } }
  }

  if (!cleaned.length){
    showImportedState('This saved number set has no valid main numbers. Choose another set below or select manually.', true);
    setStatus('err', 'Could not load the saved number set. Choose another set below or select numbers manually.');
    return false;
  }

  importedPayload = {
    play_set_id: toPositiveInt(p.play_set_id),
    lottery_id: toPositiveInt(p.lottery_id),
    target_game_id: String(p.target_game_id || ''),
    target_prediction_type: String(p.target_prediction_type || ''),
    target_pick_size: toPositiveInt(p.target_pick_size),
    target_extra_count: toPositiveInt(p.target_extra_count),
    main_numbers: cleaned,
    extra_ball_numbers: extras,
    label: String(p.label || ''),
    origin_source: String(p.origin_source || '')
  };

  if (hasZero) {
    includeZero = true;
    if (dom.zeroSel) { dom.zeroSel.value = '1'; }
  }
  if (maxSeen > maxNum) {
    maxNum = maxSeen > 99 ? 99 : maxSeen;
    renderMaxDropdown();
    if (dom.maxSel) { dom.maxSel.value = String(maxNum); }
  }

  if (wheel){
    var req = parseInt(wheel.requiredSelections, 10) || 0;
    selected = [];
    for (i = 0; i < importedPayload.main_numbers.length; i++){
      var v = importedPayload.main_numbers[i];
      if (!includeZero && v === 0) { continue; }
      if (v > maxNum) { continue; }
      selected.push(pad2(v));
      if (req > 0 && selected.length >= req) { break; }
    }
    syncGridPressed();
    updateSelectionUI();
  } else {
    selected = [];
    for (i = 0; i < importedPayload.main_numbers.length; i++){
      selected.push(pad2(importedPayload.main_numbers[i]));
    }
    updateSelectionUI();
  }

  // Build the import status message with clear next-action guidance
  var req2 = wheel ? (parseInt(wheel.requiredSelections, 10) || 0) : 0;
  var have2 = importedPayload.main_numbers.length;
  var diff2 = req2 > 0 ? (req2 - have2) : 0;
  var actionHint = '';
  if (req2 > 0 && diff2 > 0) {
    actionHint = ' Add ' + diff2 + ' more number' + (diff2 !== 1 ? 's' : '') + ' from the grid below to continue.';
  } else if (req2 > 0 && diff2 < 0) {
    var extra2 = -diff2;
    actionHint = ' Remove ' + extra2 + ' number' + (extra2 !== 1 ? 's' : '') + ' from the grid below to continue.';
  } else if (req2 > 0 && diff2 === 0) {
    actionHint = ' Your set is ready. Click Generate Combinations.';
  }
  var setName = importedPayload.label ? importedPayload.label : ('Play Set #' + (importedPayload.play_set_id || ''));
  showImportedState(
    'Loaded: ' + setName + ' - ' + have2 + ' number' + (have2 !== 1 ? 's' : '') + '.' + actionHint,
    false
  );
  // Hide the picker since a set is now loaded
  if (dom.pickerSection) { dom.pickerSection.style.display = 'none'; }
  return true;
}

function importFromQueryParams(){
  var endpointParam = qs('play_set_endpoint');
  var fallbackMain = qs('main_numbers');
  var fallbackLabel = qs('label');
  var reqUrl;

  playSetId = toPositiveInt(qs('play_set_id'));
  if (endpointParam && String(endpointParam).trim()) {
    playSetEndpoint = String(endpointParam).trim();
  }

  if (playSetId > 0){
    reqUrl = playSetEndpoint + (playSetEndpoint.indexOf('?') === -1 ? '?' : '&') + 'play_set_id=' + encodeURIComponent(String(playSetId));
    setStatus('ok', 'Loading imported play set #' + playSetId + ' ...');
    return fetch(reqUrl, { credentials: 'same-origin', cache: 'no-store' })
      .then(function(r){
        if (!r.ok) { throw new Error('HTTP ' + r.status + ' while loading play set payload.'); }
        return r.json();
      })
      .then(function(j){
        if (!j || !j.ok || !j.payload) { throw new Error((j && j.error) ? j.error : 'Play set payload was not returned.'); }
        if (!applyImportedPayload(j.payload)) { throw new Error('Play set payload failed validation.'); }
        return true;
      })
      .catch(function(err){
        showImportedState('Could not load the saved number set. ' + String(err && err.message ? err.message : err), true);
        setStatus('err', 'Could not load the saved number set. Choose from the list below or select manually.');
        return false;
      });
  }

  if (fallbackMain){
    applyImportedPayload({
      main_numbers: parseCsvNumbers(fallbackMain),
      extra_ball_numbers: parseCsvNumbers(qs('extra_ball_numbers')),
      label: fallbackLabel || '',
      origin_source: qs('origin_source') || 'imported',
      target_game_id: qs('target_game_id') || '',
      target_prediction_type: qs('target_prediction_type') || '',
      target_pick_size: toPositiveInt(qs('target_pick_size')),
      target_extra_count: toPositiveInt(qs('target_extra_count'))
    });
    return Promise.resolve(true);
  }

  return Promise.resolve(false);
}

function onPick(e){
  if (!wheel) { return; }

  var v = e.currentTarget.getAttribute('data-v');
  var req = parseInt(wheel.requiredSelections,10) || 0;
  var idx = selected.indexOf(v);

  if (idx === -1){
    if (selected.length >= req){
      setStatus('err', 'You already selected the required ' + req + ' numbers.');
      return;
    }
    selected.push(v);
  } else {
    selected.splice(idx, 1);
  }

  syncGridPressed();
  updateSelectionUI();
  setStatus('ok', 'Selected ' + selected.length + ' of ' + req + '.');
}

function randomSelect(){
  if (!wheel) { return; }
  resetSelection();

  var req = parseInt(wheel.requiredSelections,10) || 0;
  var pool = [];
  var start = includeZero ? 0 : 1;

  for (var i = start; i <= maxNum; i++){ pool.push(pad2(i)); }

  for (var j = pool.length - 1; j > 0; j--){
    var k = Math.floor(Math.random() * (j + 1));
    var tmp = pool[j];
    pool[j] = pool[k];
    pool[k] = tmp;
  }

  selected = pool.slice(0, req);
  syncGridPressed();
  updateSelectionUI();
  setStatus('ok', 'Randomly selected ' + req + ' numbers.');
}

function createWheel(){
  if (!wheel) { return; }

  var req = parseInt(wheel.requiredSelections, 10) || 0;
  if (selected.length !== req){
    setStatus('err', 'Select exactly ' + req + ' numbers first.');
    return;
  }

  var lines = (wheel.wheelLines && Array.isArray(wheel.wheelLines)) ? wheel.wheelLines :
              (wheel.lines && Array.isArray(wheel.lines)) ? wheel.lines : [];

  if (!lines.length){
    setStatus('err', 'This wheel system has no line template. Verify wheels.json contains "wheelLines" or "lines" for system ' + (systemId || '') + '.');
    return;
  }

  builtLines = [];
  builtCombos = [];

  for (var i = 0; i < lines.length; i++){
    var idxs = lines[i];
    if (!idxs || !idxs.length || !Array.isArray(idxs)) { continue; }

    var vals = [];
    for (var j = 0; j < idxs.length; j++){
      var n = parseInt(idxs[j], 10) || 0;
      if (n <= 0) { continue; }
      var v = selected[n - 1];
      if (typeof v === 'string' && v) { vals.push(v); }
    }

    var pick = parseInt(wheel.pick, 10) || parseInt(wheel.pickSize, 10) || 0;
    if (pick > 0 && vals.length !== pick) { continue; }

    builtCombos.push(vals);
    builtLines.push(vals.join(' - '));
  }

  if (!builtLines.length){
    setStatus('err', 'No valid combinations could be generated. This usually means the line template indices do not match the selected pool size.');
    if (dom.outBody) { dom.outBody.innerHTML = ''; }
    updateSelectionUI();
    return;
  }

  if (dom.outBody){
    dom.outBody.innerHTML = '';
    for (var r = 0; r < builtLines.length; r++){
      var tr = document.createElement('tr');
      var td1 = document.createElement('td');
      var td2 = document.createElement('td');
      td1.textContent = String(r + 1);
      td2.textContent = builtLines[r];
      tr.appendChild(td1);
      tr.appendChild(td2);
      dom.outBody.appendChild(tr);
    }
  }

  try { renderCoverageDiagnostics(); } catch (e) {}

  updateSelectionUI();
  syncPrintHeader();
  setStatus('ok', 'Wheel created: ' + builtLines.length + ' lines.');
}

function renderCoverageDiagnostics(){
  var cov = document.getElementById('leCoverage');
  if (!cov) { return; }
  if (!wheel || !builtCombos || !builtCombos.length){ cov.innerHTML = ''; return; }

  var freq = {};
  var pairFreq = {};
  var totalPairsPossible = 0;

  for (var i = 0; i < selected.length; i++){ freq[selected[i]] = 0; }
  if (selected.length >= 2){ totalPairsPossible = (selected.length * (selected.length - 1)) / 2; }

  for (var r = 0; r < builtCombos.length; r++){
    var line = builtCombos[r] || [];

    for (var a = 0; a < line.length; a++){
      var v = line[a];
      if (freq[v] == null) { freq[v] = 0; }
      freq[v] += 1;
    }

    for (var x = 0; x < line.length; x++){
      for (var y = x + 1; y < line.length; y++){
        var p1 = line[x];
        var p2 = line[y];
        var key = (p1 < p2) ? (p1 + '|' + p2) : (p2 + '|' + p1);
        if (!pairFreq[key]) { pairFreq[key] = 0; }
        pairFreq[key] += 1;
      }
    }
  }

  var maxF = 0, minF = 999999, sumF = 0;
  for (var k in freq){
    if (!Object.prototype.hasOwnProperty.call(freq, k)) { continue; }
    var f = freq[k] || 0;
    if (f > maxF) { maxF = f; }
    if (f < minF) { minF = f; }
    sumF += f;
  }

  var keys = Object.keys(freq);
  var avgF = (keys.length > 0) ? (sumF / keys.length) : 0;
  var pairsCovered = Object.keys(pairFreq).length;
  var pairCoveragePct = (totalPairsPossible > 0) ? Math.round((pairsCovered / totalPairsPossible) * 100) : 0;

  var pairList = [];
  for (var pk in pairFreq){
    if (!Object.prototype.hasOwnProperty.call(pairFreq, pk)) { continue; }
    pairList.push({ key: pk, count: pairFreq[pk] });
  }
  pairList.sort(function(a1, b1){ return a1.count - b1.count; });
  var under = pairList.slice(0, 10);

  var LT = String.fromCharCode(60);
  var GT = String.fromCharCode(62);
  var html = '';

  html += LT + 'h3' + GT + 'Coverage Diagnostics' + LT + '/h3' + GT;
  html += LT + 'div class="le-cov-grid"' + GT;
  html += LT + 'div class="le-cov-kpi"' + GT;
  html += LT + 'div class="k"' + GT + 'Single-number coverage' + LT + '/div' + GT;
  html += LT + 'div class="v"' + GT + 'Min ' + minF + ' | Avg ' + avgF.toFixed(1) + ' | Max ' + maxF + LT + '/div' + GT;
  html += LT + '/div' + GT;
  html += LT + 'div class="le-cov-kpi"' + GT;
  html += LT + 'div class="k"' + GT + 'Pair coverage' + LT + '/div' + GT;
  html += LT + 'div class="v"' + GT + pairsCovered + ' / ' + totalPairsPossible + ' pairs (' + pairCoveragePct + '%)' + LT + '/div' + GT;
  html += LT + '/div' + GT;
  html += LT + '/div' + GT;

  var show = selected.slice(0, 20);
  html += LT + 'div class="le-cov-bars"' + GT;
  for (var i2 = 0; i2 < show.length; i2++){
    var num = show[i2];
    var f2 = freq[num] || 0;
    var pct = (maxF > 0) ? Math.round((f2 / maxF) * 100) : 0;

    html += LT + 'div class="le-cov-row"' + GT;
    html += LT + 'div class="le-cov-tag"' + GT + num + LT + '/div' + GT;
    html += LT + 'div class="le-cov-bar"' + GT + LT + 'span style="width:' + pct + '%"' + GT + LT + '/span' + GT + LT + '/div' + GT;
    html += LT + 'div class="le-cov-val"' + GT + f2 + LT + '/div' + GT;
    html += LT + '/div' + GT;
  }
  html += LT + '/div' + GT;

  if (under.length){
    html += LT + 'div class="le-cov-small"' + GT + LT + 'strong' + GT + 'Least-covered pairs' + LT + '/strong' + GT + ' (lowest appearances across lines):' + LT + 'br' + GT;
    for (var u = 0; u < under.length; u++){
      var parts = under[u].key.split('|');
      html += parts[0] + '-' + parts[1] + ' (' + under[u].count + ')' + (u < under.length - 1 ? ', ' : '');
    }
    html += LT + '/div' + GT;
  }

  html += LT + 'div class="le-cov-small"' + GT +
          'SKAI-style note: These diagnostics summarize how evenly your wheel distributes coverage across your selected pool. ' +
          'This describes structure and balance; it does not guarantee results.' +
          LT + '/div' + GT;

  cov.innerHTML = html;
}

function downloadTxt(){
  if (!builtLines || !builtLines.length){ setStatus('err', 'Create the wheel before downloading.'); return; }
  var txt = builtLines.join('\r\n') + '\r\n';
  var blob = new Blob([txt], {type:'text/plain;charset=utf-8'});
  var a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = (systemId ? systemId : 'wheel') + '-combinations.txt';
  document.body.appendChild(a);
  a.click();
  setTimeout(function(){
    try{ URL.revokeObjectURL(a.href); }catch(_){}
    try{ a.parentNode.removeChild(a); }catch(_){}
  }, 200);
}

  var txt = buildCopyText();

  if (navigator.clipboard && navigator.clipboard.writeText){

      setStatus('ok', 'Wheel copied.');
    }).catch(function(){ fallbackCopy(txt); });
  } else {

  }
}
function fallbackCopy(txt){
  try{
    var ta = document.createElement('textarea');
    ta.value = txt;
    ta.setAttribute('readonly', 'readonly');
    ta.style.position = 'fixed';
    ta.style.left = '-9999px';
    document.body.appendChild(ta);
    ta.select();
    ta.setSelectionRange(0, ta.value.length);

    if (document.execCommand('copy')) {
      flashCopiedState();
      setStatus('ok', 'Wheel copied.');
    } else {
      setStatus('err', 'Copy failed. Please use Download instead.');
    }
    document.body.removeChild(ta);
  }catch(e){
    setStatus('err', 'Copy failed. Please use Download instead.');
  }
}

function printWheel(){
  if (!builtLines || !builtLines.length){ setStatus('err', 'Create the wheel before printing.'); return; }

  var txt = buildCopyText();

  var win = window.open('', '_blank');
  if (!win){
    setStatus('err', 'Print window was blocked. Please allow pop-ups for this site and try again.');
    return;
  }
  win.document.body.style.fontFamily = 'monospace';
  win.document.body.style.fontSize = '12pt';
  win.document.body.style.whiteSpace = 'pre-wrap';
  win.document.body.textContent = txt;
  win.focus();
  setStatus('ok', 'Print dialog opened.');
  win.print();
}

function loadWheelFromJson(sysId){
  function normalizeWheel(raw){
    var w = raw || {};
    var pick = parseInt(w.pick, 10);
    var required = parseInt(w.requiredSelections, 10);
    var lines = Array.isArray(w.lines) ? w.lines : (Array.isArray(w.wheelLines) ? w.wheelLines : []);
    var max = parseInt(w.max, 10);
    var assuranceN = parseInt(w.assurance, 10);
    var assuranceLabel;
    var cc;

    if (isNaN(pick)) { pick = parseInt(w.pickSize, 10); }
    if (isNaN(pick)) { pick = 5; }

    if (isNaN(required) || required <= 0){
      if (isNaN(assuranceN)) { assuranceN = 2; }
      required = pick + assuranceN;
    }

    if (isNaN(max) || max <= 0) { max = 99; }

    assuranceLabel = (typeof w.assuranceText === 'string' && w.assuranceText.trim()) ?
      w.assuranceText.trim() :
      (!isNaN(assuranceN) ? ('Assurance ' + assuranceN) : '-');

    cc = parseInt(w.combinationsCount, 10);

    return {
      name:(w.name ? String(w.name) : ''),
      pick:pick,
      requiredSelections:required,
      max:max,
      assuranceLabel:assuranceLabel,
      lines:lines,
      combinationsCount:(!isNaN(cc) ? cc : lines.length)
    };
  }

  function applyWheel(sysIdLocal, rawWheel){
    var norm = normalizeWheel(rawWheel);
    wheel = norm;
    systemId = sysIdLocal;

    if (dom.sys) { dom.sys.textContent = sysIdLocal; }
    if (dom.pick) { dom.pick.textContent = String(norm.pick); }
    if (dom.pool) { dom.pool.textContent = String(norm.requiredSelections); }
    if (dom.req) { dom.req.textContent = String(norm.requiredSelections); }
    if (dom.assurance) { dom.assurance.textContent = String(norm.assuranceLabel); }
    if (dom.lines) { dom.lines.textContent = String(norm.combinationsCount); }

    if (dom.title){ dom.title.textContent = norm.name ? norm.name : ('Wheel Builder | ' + sysIdLocal); }
    if (dom.lead){ dom.lead.textContent = 'Your wheel is active. Choose your ' + norm.requiredSelections + ' numbers below, then click Generate Combinations.'; }

    var titleText = (norm.name ? norm.name : ('Wheel ' + sysIdLocal)) + ' | Wheel Builder | LottoExpert';
    var descriptionText = 'Use the LottoExpert wheel builder to generate combinations for system ' + sysIdLocal + '. Select ' + norm.requiredSelections + ' numbers, review coverage diagnostics, copy, print, or download your wheel output.';
    applyBuilderSeo(titleText, descriptionText);

    maxNum = norm.max;
    renderMaxDropdown();
    if (dom.maxSel) { dom.maxSel.value = String(maxNum); }
    includeZero = false;
    if (dom.zeroSel) { dom.zeroSel.value = '0'; }

    resetSelection();
    renderGrid();
    if (importedPayload && importedPayload.main_numbers && importedPayload.main_numbers.length){
      applyImportedPayload(importedPayload);
    }
    updateSelectionUI();
    syncPrintHeader();
    if (importedPayload && importedPayload.main_numbers && importedPayload.main_numbers.length){
      var req3 = norm.requiredSelections;
      var have3 = selected.length;
      var diff3 = req3 - have3;
      if (diff3 === 0) {
        setStatus('ok', 'Wheel active. Your number set is ready. Click Generate Combinations.');
      } else if (diff3 > 0) {
        setStatus('ok', 'Wheel active. Your set loaded. Add ' + diff3 + ' more number' + (diff3 !== 1 ? 's' : '') + ' below to continue.');
      } else {
        setStatus('ok', 'Wheel active. Your set loaded. Remove ' + (-diff3) + ' number' + ((-diff3) !== 1 ? 's' : '') + ' below to continue.');
      }
    } else {
      setStatus('ok', 'Wheel active. Select ' + norm.requiredSelections + ' numbers below to generate combinations.');
    }
  }

  function fallbackFetch(){
    setStatus('ok', 'Loading wheel system ' + sysId + '... (fallback mode)');
    fetch(WHEELS_JSON_URL, {cache:'no-store'})
      .then(function(r){
        if (!r.ok){ throw new Error('HTTP ' + r.status + ' while loading wheels.json'); }
        return r.json();
      })
      .then(function(j){
        if (!j || !j.wheels || !j.wheels[sysId]){
          throw new Error('system_id not found in wheels.json: ' + sysId);
        }
        applyWheel(sysId, j.wheels[sysId]);
      })
      .catch(function(err){
        setStatus('err', 'Wheel load failed: ' + String(err && err.message ? err.message : err));
        if (dom.lead) {
          dom.lead.textContent = 'Could not load the wheel library. Verify wheels.json URL and server permissions.';
        }
      });
  }

  var workerSrc = [
    "self.onmessage=function(e){",
    "var url=e.data.url,id=e.data.id;",
    "fetch(url,{cache:'no-store'}).then(function(r){",
    "if(!r.ok){throw new Error('HTTP '+r.status+' while loading wheels.json');}",
    "return r.text();",
    "}).then(function(t){",
    "var j=JSON.parse(t);",
    "if(!j||!j.wheels||!j.wheels[id]){self.postMessage({ok:false,err:'system_id not found: '+id});return;}",
    "self.postMessage({ok:true,wheel:j.wheels[id]});",
    "}).catch(function(err){",
    "self.postMessage({ok:false,err:String(err&&err.message?err.message:err)});",
    "});",
    "};"
  ].join(" ");

  try{
    var blob = new Blob([workerSrc], {type:'application/javascript'});
    var worker = new Worker(URL.createObjectURL(blob));
    worker.onmessage = function(ev){
      var msg = ev.data || {};
      if (!msg.ok){
        setStatus('err', 'Worker load failed: ' + (msg.err || 'Unknown error') + ' (trying fallback...)');
        fallbackFetch();
        return;
      }
      applyWheel(sysId, msg.wheel);
    };
    worker.onerror = function(){ fallbackFetch(); };
    worker.postMessage({url: WHEELS_JSON_URL, id: sysId});
  }catch(e){
    fallbackFetch();
  }
}

function init(){
  systemId = qs('system_id');
  var listEndpointParam = qs('play_sets_endpoint');
  if (listEndpointParam && String(listEndpointParam).trim()) {
    playSetListEndpoint = String(listEndpointParam).trim();
  }
  renderMaxDropdown();
  wireButtons();
  applyBuilderSeo('Wheel Builder | LottoExpert', 'Use the LottoExpert wheel builder to generate combinations from a selected wheel system.');
  if (dom.systemInput && systemId) { dom.systemInput.value = systemId; }

  importFromQueryParams().then(function(importedOk){
    if (!systemId){
      if (importedOk){
        if (dom.lead) {
          dom.lead.textContent = 'Your number set is loaded. Choose a wheel to apply it to, then generate combinations.';
        }
        if (dom.sys) { dom.sys.textContent = '-'; }
        setStatus('ok', 'Number set loaded. Enter a wheel ID in the field below or go to the catalog to choose a wheel.');
        fetchPlaySets();
        return;
      }
      setStatus('err', 'No wheel selected. Go back to the catalog and choose a wheel to continue.');
      if (dom.lead) { dom.lead.textContent = 'Choose a wheel from the catalog first, then come back here to generate combinations.'; }
      fetchPlaySets();
      return;
    }
    loadWheelFromJson(systemId);
    if (!importedOk) {
      fetchPlaySets();
    }
  });
}

if (document.readyState === 'loading'){
  document.addEventListener('DOMContentLoaded', init);
}else{
  init();
}
